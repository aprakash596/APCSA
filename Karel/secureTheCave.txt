void secureTheCave()
{
    turnLeft();
    grabAndDrop();
}
void grabAndDrop()
{
    if(frontIsClear())
    {
        moveForward();
        if(onBeeper())
        {
            pickBeeper();
            grabAndDrop();
            dropBeeper();
            moveForward();
            grabAndDrop();
        }
    }
    else
    {
        turnAround();
    }
}
