void defuseTwoBombs()
{
    moveForward();
    checkForWall();
    turnAround();
    checkForWall();
    turnRight();
    checkForWall();
    turnAround();
    checkForWall();
    turnAround();
}
void checkForWall()
{
    while(frontIsClear())
    {
        moveForward();
        if(onBeeper())
        {
            pickBeeper();
        }
        
    }
}
