void solveTheMaze()
{
    while(! onBeeper())
    {
        if(! leftIsClear() && ! frontIsClear() && ! rightIsClear())
        {
            turnAround();
        }
        else 
        {
            if(! leftIsClear() && frontIsClear())
            {
                moveForward();
            }
            else 
            {
                if(rightIsClear() && ! frontIsClear() && ! leftIsClear())
                {
                    turnRight();
                    moveForward();
                }
                else 
                {
                    
                    turnLeft();
                    moveForward();
                    
                }
            }
        }
    }
}
