void mowTheLawn()
{
    moveForward();
    mowingRow();
    onTheRight();
    mowingRow();
    onTheLeft();
    mowingRow();
    onTheRight();
    mowingRow();
    onTheLeft();
    mowingRow();
    onTheRight();
    mowingRow();
}
void onTheRight()
{
    turnLeft();
    moveForward();
    turnLeft();
    moveForward();
}
void onTheLeft()
{
    turnRight();
    moveForward();
    turnRight();
    moveForward();
}
void mowingRow()
{
    while(onBeeper())
    {
        pickBeeper();
        moveForward();
    }
}
