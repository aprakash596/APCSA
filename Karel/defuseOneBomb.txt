void defuseOneBomb()
{
    moveForward();
    repeat(8)
    {
        checkBomb();
    }
    turnAround();
    repeat(9)
    {
        moveForward();
    }
}

void checkBomb()
{
    moveForward();
    if(onBeeper())
    {
        pickBeeper();
    }
}
