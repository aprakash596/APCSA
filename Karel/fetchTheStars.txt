void fetchTheStars()
{
    repeat(10)
    {
        turnLeft();
        repeatForward();
        dropBeeper();
        turnLeft();
        if(frontIsClear())
        {
            moveForward();
        }
    }
}
void repeatForward()
{
    if(frontIsClear())
    {
        moveForward();
        repeatForward();
        moveForward();
    }
    else
    {
        pickBeeper();
        turnAround();
    }
}
