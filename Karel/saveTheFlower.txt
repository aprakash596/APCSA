void saveTheFlower()
{
    moveForward();
    pickBeeper();
    while(! frontIsClear())
    {
        scalingTheFlower();
    }
    dropBeeper();
    
    while(frontIsClear())
    {
        scalingDownFlower();
    }
}
void scalingTheFlower()
{
    turnLeft();
    moveForward();
    moveForward();
    turnRight();
    moveForward();
}

void scalingDownFlower()
{
    moveForward();
    turnRight();
    moveForward();
    moveForward();
    turnLeft();
}
